# UR5 机械臂抓取与放置案例

## 场景描述

使用 UR5 六轴工业机器人在 CoppeliaSim 中实现**颜色分拣**抓取与放置（Pick and Place）任务。机器人根据视觉检测结果将不同颜色的工件抓取到对应位置。

## 场景搭建

### 所需模型
1. **UR5** — 从 `models/robots/non-mobile/UR5.ttm` 导入
2. **RG2 夹爪** — 安装在 UR5 末端
3. **工件** — 不同颜色的立方体（Red, Blue, Green）
4. **放置区** — 对应颜色的目标位置
5. **视觉传感器** — 检测工件颜色和位置

### 模型层级结构

## 核心实现 (Lua 子脚本)

### FSM 状态机
```lua
-- 状态定义
STATE = {
    IDLE = 0,
    APPROACH = 1,
    GRASP = 2,
    LIFT = 3,
    TRANSFER = 4,
    PLACE = 5,
    RELEASE = 6,
    RETREAT = 7
}

function sysCall_init()
    simIK = require('simIK')
    
    -- 获取对象句柄
    joints = {}
    for i = 1, 6 do
        joints[i] = sim.getObject('/UR5/joint' .. i)
    end
    tip = sim.getObject('/UR5/tip')
    target = sim.getObject('/UR5/target')
    
    -- 夹爪
    gripperLeft = sim.getObject('/RG2/RG2_left')
    gripperRight = sim.getObject('/RG2/RG2_right')
    
    -- 初始化 IK
    setupIK()
    
    -- 状态变量
    currentState = STATE.IDLE
    graspPos = {0.3, 0.0, 0.1}
    placePos = {0.3, 0.4, 0.1}
    approachOffset = {0, 0, 0.15}  -- 接近偏移量
end

function setupIK()
    ikEnv = simIK.createEnvironment()
    ikGroup = simIK.createGroup(ikEnv)
    
    base = sim.getObject('/UR5')
    ikElement, simToIkMap, ikToSimMap = simIK.addElementFromScene(
        ikEnv, ikGroup, base, tip, target,
        simIK.constraint_pose
    )
    
    simIK.setGroupCalculation(ikEnv, ikGroup,
        simIK.method_pseudo_inverse,
        simIK.constraint_pose)
end

function setGripper(open)
    if open then
        -- 打开夹爪
        sim.setJointPosition(gripperLeft, 0.0)
        sim.setJointPosition(gripperRight, 0.0)
    else
        -- 闭合夹爪
        sim.setJointPosition(gripperLeft, 0.045)
        sim.setJointPosition(gripperRight, -0.045)
    end
end

function moveToPosition(pos, orient)
    sim.setObjectPosition(target, -1, pos)
    if orient then
        sim.setObjectOrientation(target, -1, orient)
    end
    
    local result = simIK.handleGroup(ikEnv, ikGroup, {
        syncWorlds = true,
        syncStep = true
    })
    
    if result ~= simIK.result_success then
        print("IK 求解失败")
        return false
    end
    
    -- 等待到达
    sim.wait(0.5)
    return true
end

function sysCall_actuation()
    -- 状态机逻辑
    if currentState == STATE.IDLE then
        -- 检测是否有新工件
        if detectNewPart() then
            currentState = STATE.APPROACH
        end
        
    elseif currentState == STATE.APPROACH then
        -- 接近工件（上方偏移）
        local approachPos = {
            graspPos[1] + approachOffset[1],
            graspPos[2] + approachOffset[2],
            graspPos[3] + approachOffset[3]
        }
        if moveToPosition(approachPos, {math.pi, 0, 0}) then
            setGripper(true)
            currentState = STATE.GRASP
        end
        
    elseif currentState == STATE.GRASP then
        -- 移动到抓取位置并闭合夹爪
        if moveToPosition(graspPos, {math.pi, 0, 0}) then
            setGripper(false)
            sim.wait(0.3)
            currentState = STATE.LIFT
        end
        
    elseif currentState == STATE.LIFT then
        -- 提起工件
        local liftPos = {
            graspPos[1], graspPos[2], graspPos[3] + 0.2
        }
        if moveToPosition(liftPos) then
            currentState = STATE.TRANSFER
        end
        
    elseif currentState == STATE.TRANSFER then
        -- 移动到放置区上方
        local abovePlace = {
            placePos[1], placePos[2], placePos[3] + 0.2
        }
        if moveToPosition(abovePlace) then
            currentState = STATE.PLACE
        end
        
    elseif currentState == STATE.PLACE then
        -- 下降到放置位置
        if moveToPosition(placePos, {math.pi, 0, 0}) then
            setGripper(true)
            sim.wait(0.3)
            currentState = STATE.RETREAT
        end
        
    elseif currentState == STATE.RETREAT then
        -- 退回安全位置
        local retreatPos = {0.0, -0.3, 0.4}
        if moveToPosition(retreatPos) then
            currentState = STATE.IDLE
            print("完成一次抓取放置循环")
        end
    end
end

function detectNewPart()
    -- 使用接近传感器或视觉传感器检测工件
    local result, dist, point = sim.readProximitySensor(sensorHandle)
    if result > 0 then
        -- 获取工件位置
        local partPos = sim.getObjectPosition(partHandle, -1)
        graspPos = partPos
        return true
    end
    return false
end
```

## Python 外部控制版本

```python
from coppeliasim_zmqremoteapi_client import RemoteAPIClient
import time
import math

class UR5PickAndPlace:
    def __init__(self):
        self.client = RemoteAPIClient()
        self.sim = self.client.getObject('sim')
        self.simIK = self.client.getObject('simIK')
        
    def setup(self):
        # 获取关节句柄
        self.joints = [self.sim.getObject(f'/UR5/joint{i}') for i in range(1, 7)]
        self.tip = self.sim.getObject('/UR5/tip')
        self.target = self.sim.getObject('/UR5/target')
        self.base = self.sim.getObject('/UR5')
        
        # 设置 IK
        self.ik_env = self.simIK.createEnvironment()
        self.ik_group = self.simIK.createGroup(self.ik_env)
        
        self.simIK.addElementFromScene(
            self.ik_env, self.ik_group,
            self.base, self.tip, self.target,
            self.simIK.constraint_pose
        )
        
    def move_to(self, position, orientation=None):
        self.sim.setObjectPosition(self.target, -1, position)
        if orientation:
            self.sim.setObjectOrientation(self.target, -1, orientation)
        
        result = self.simIK.handleGroup(self.ik_env, self.ik_group,
                                         {'syncWorlds': True, 'syncStep': True})
        return result == self.simIK.result_success
    
    def pick_and_place(self, grasp_pos, place_pos):
        """执行一次完整的 pick-and-place"""
        approach = [grasp_pos[0], grasp_pos[1], grasp_pos[2] + 0.15]
        lift = [grasp_pos[0], grasp_pos[1], grasp_pos[2] + 0.20]
        above_place = [place_pos[0], place_pos[1], place_pos[2] + 0.15]
        orient = [math.pi, 0, 0]
        
        steps = [
            ("Approach", approach, orient),
            ("Grasp", grasp_pos, orient),
            ("Lift", lift, orient),
            ("Transfer", above_place, orient),
            ("Place", place_pos, orient),
            ("Retreat", [0.0, -0.3, 0.4], [0, 0, 0])
        ]
        
        for name, pos, ori in steps:
            print(f"执行: {name}")
            if not self.move_to(pos, ori):
                print(f"失败: {name}")
                return False
            time.sleep(0.3)
        
        return True

# 使用
robot = UR5PickAndPlace()
robot.setup()
robot.sim.startSimulation()
time.sleep(0.5)

robot.pick_and_place(
    grasp_pos=[0.3, 0.0, 0.1],
    place_pos=[-0.2, 0.4, 0.05]
)

robot.sim.stopSimulation()
```

## 关键经验总结

### 1. IK 配置要点
- 使用 `simIK.addElementFromScene()` 自动解析 UR5 运动链
- 设置合适的 `method`（伪逆法通用性最好）
- 检查关节限位是否合理

### 2. 抓取策略
- 先移动到工件上方（approach），再垂直下降抓取
- 提起（lift）后再水平移动，避免碰撞
- 放置时同样垂直下降

### 3. 碰撞避免
- 在场景中添加适当的碰撞检测对
- 使用 OMPL 规划复杂环境下的避障路径
- 注意工作台和其他物体的碰撞边界

### 4. 时序控制
```lua
-- 等待夹爪动作完成
sim.wait(0.3)

-- 等待关节到达位置
local posDiff = math.abs(sim.getJointPosition(joint) - targetPos)
while posDiff > 0.01 do
    posDiff = math.abs(sim.getJointPosition(joint) - targetPos)
    sim.switchThread()
end
```
